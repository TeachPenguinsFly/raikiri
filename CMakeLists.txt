cmake_minimum_required(VERSION 3.28)

# VERSION holds the release, a pre-release with its suffix; cmake's own version takes only the numbers
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" RAIKIRI_VERSION LIMIT_COUNT 1)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" RAIKIRI_VERSION_CORE "${RAIKIRI_VERSION}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")

project(raikiri VERSION ${RAIKIRI_VERSION_CORE} LANGUAGES CXX)

if(PROJECT_SOURCE_DIR STREQUAL PROJECT_BINARY_DIR)
  message(FATAL_ERROR "In-source builds are not allowed. Use a CMake preset.")
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# no modules here; clang's dependency scan would also trip over the shared pch
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

# libc++ before apple clang 17 and llvm 20 hides std::stop_token behind its experimental switch
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_FLAGS "-std=c++23")
check_cxx_source_compiles("#include <stop_token>\nint main() { std::stop_source s; return s.stop_requested(); }"
  RAIKIRI_HAS_STOP_TOKEN)
if(NOT RAIKIRI_HAS_STOP_TOKEN)
  add_compile_options(-fexperimental-library)
  add_link_options(-fexperimental-library)
endif()
unset(CMAKE_REQUIRED_FLAGS)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(RAIKIRI_BUILD_TESTS "Build unit tests" ${PROJECT_IS_TOP_LEVEL})
option(RAIKIRI_BUILD_BENCH "Build benchmarks" ON)
option(RAIKIRI_WERROR "Treat warnings as errors" ON)
option(RAIKIRI_PCH "Share one precompiled header across every first-party target" ON)
set(RAIKIRI_SANITIZE "" CACHE STRING "Sanitizers, e.g. 'address,undefined' or 'thread'")
set(RAIKIRI_MARCH "" CACHE STRING "Explicit -march target (never 'native'); empty = compiler default")

include(cmake/Warnings.cmake)
include(cmake/Sanitizers.cmake)

# with RAIKIRI_PCH on, ccache needs the pch sloppiness or every tu misses
option(RAIKIRI_CCACHE "Compile through ccache when the host has it" OFF)
if(RAIKIRI_CCACHE)
  find_program(RAIKIRI_CCACHE_BIN ccache)
  if(RAIKIRI_CCACHE_BIN)
    set(CMAKE_CXX_COMPILER_LAUNCHER
        ${CMAKE_COMMAND} -E env CCACHE_SLOPPINESS=pch_defines,time_macros ${RAIKIRI_CCACHE_BIN})
    message(STATUS "ccache: ${RAIKIRI_CCACHE_BIN}")
  endif()
endif()

if(RAIKIRI_MARCH)
  if(RAIKIRI_MARCH STREQUAL "native")
    message(FATAL_ERROR "-march=native is banned: build host != deploy host causes SIGILL. Name the target explicitly.")
  endif()
  add_compile_options(-march=${RAIKIRI_MARCH})
endif()
